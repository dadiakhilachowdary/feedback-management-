/* =====================================================
   STORAGE
===================================================== */

let users =
JSON.parse(
    localStorage.getItem("users")
) || [];


let feedbacks =
JSON.parse(
    localStorage.getItem("feedbacks")
) || [];


/* =====================================================
   PAGE NAVIGATION
===================================================== */

function showPage(page){

    const pages = [
        "homePage",
        "registerPage",
        "userLoginPage",
        "feedbackPage",
        "reviewsPage",
        "adminLoginPage",
        "adminPage"
    ];

    pages.forEach(function(id){

        document
        .getElementById(id)
        .classList
        .add("hidden");

    });


    if(page === "home"){

        document
        .getElementById("homePage")
        .classList
        .remove("hidden");

        updateHome();

    }


    if(page === "register"){

        document
        .getElementById("registerPage")
        .classList
        .remove("hidden");

    }


    if(page === "userLogin"){

        document
        .getElementById("userLoginPage")
        .classList
        .remove("hidden");

    }


    if(page === "feedback"){

        let loggedUser =
        JSON.parse(
            localStorage.getItem("loggedUser")
        );


        if(!loggedUser){

            alert(
                "Please register and login first!"
            );

            showPage("userLogin");

            return;

        }


        document
        .getElementById("feedbackPage")
        .classList
        .remove("hidden");

        loadLoggedUser();

    }


    if(page === "reviews"){

        document
        .getElementById("reviewsPage")
        .classList
        .remove("hidden");

        displayReviews();

    }


    if(page === "adminLogin"){

        document
        .getElementById("adminLoginPage")
        .classList
        .remove("hidden");

    }


    if(page === "admin"){

        let admin =
        sessionStorage.getItem(
            "adminLoggedIn"
        );


        if(admin !== "true"){

            showPage("adminLogin");

            return;

        }


        document
        .getElementById("adminPage")
        .classList
        .remove("hidden");

        updateAdminDashboard();

    }


    window.scrollTo({
        top:0,
        behavior:"instant"
    });

}


/* =====================================================
   REGISTER
===================================================== */

document
.getElementById("registerForm")
.addEventListener(
    "submit",
    function(e){

        e.preventDefault();


        let name =
        document
        .getElementById("registerName")
        .value.trim();


        let email =
        document
        .getElementById("registerEmail")
        .value.trim()
        .toLowerCase();


        let phone =
        document
        .getElementById("registerPhone")
        .value.trim();


        let password =
        document
        .getElementById("registerPassword")
        .value;


        let confirmPassword =
        document
        .getElementById("confirmPassword")
        .value;


        if(password !== confirmPassword){

            alert(
                "Passwords do not match!"
            );

            return;

        }


        let existingUser =
        users.find(
            user =>
            user.email === email
        );


        if(existingUser){

            alert(
                "This email is already registered!"
            );

            return;

        }


        let newUser = {

            id:Date.now(),

            name:name,

            email:email,

            phone:phone,

            password:password

        };


        users.push(newUser);


        localStorage.setItem(
            "users",
            JSON.stringify(users)
        );


        showNotification(
            "Registration successful!"
        );


        document
        .getElementById("registerForm")
        .reset();


        setTimeout(
            function(){

                showPage("userLogin");

            },
            1000
        );

    }
);

/* =====================================================
   LOAD USER
===================================================== */

function loadLoggedUser(){

    let user =
    JSON.parse(
        localStorage.getItem("loggedUser")
    );


    if(user){

        document
        .getElementById("feedbackName")
        .value =
        user.name;


        document
        .getElementById("feedbackEmail")
        .value =
        user.email;

    }

}


/* =====================================================
   STAR RATING
===================================================== */

let stars =
document.querySelectorAll(".star");


stars.forEach(function(star){

    star.addEventListener(
        "click",
        function(){

            let rating =
            parseInt(
                this.getAttribute(
                    "data-value"
                )
            );


            document
            .getElementById("feedbackRating")
            .value =
            rating;


            stars.forEach(function(s){

                let value =
                parseInt(
                    s.getAttribute(
                        "data-value"
                    )
                );


                if(value <= rating){

                    s.classList.add(
                        "active"
                    );

                }
                else{

                    s.classList.remove(
                        "active"
                    );

                }

            });

        }
    );

});


/* =====================================================
   SUBMIT FEEDBACK
===================================================== */

document
.getElementById("feedbackForm")
.addEventListener(
    "submit",
    function(e){

        e.preventDefault();


        let rating =
        parseInt(
            document
            .getElementById(
                "feedbackRating"
            )
            .value
        );


        if(rating === 0){

            alert(
                "Please select a rating!"
            );

            return;

        }


        let newFeedback = {

            id:Date.now(),

            name:
            document
            .getElementById(
                "feedbackName"
            )
            .value,

            email:
            document
            .getElementById(
                "feedbackEmail"
            )
            .value,

            category:
            document
            .getElementById(
                "feedbackCategory"
            )
            .value,

            rating:rating,

            message:
            document
            .getElementById(
                "feedbackMessage"
            )
            .value,

            date:
            new Date()
            .toLocaleDateString()

        };


        feedbacks.push(
            newFeedback
        );


        localStorage.setItem(
            "feedbacks",
            JSON.stringify(feedbacks)
        );


        document
        .getElementById(
            "feedbackForm"
        )
        .reset();


        document
        .getElementById(
            "feedbackRating"
        )
        .value = 0;


        stars.forEach(
            function(star){

                star.classList.remove(
                    "active"
                );

            }
        );


        showNotification(
            "Thank you! Your feedback was submitted."
        );


        setTimeout(
            function(){

                showPage("reviews");

            },
            1000
        );

    }
);


/* =====================================================
   DISPLAY REVIEWS
===================================================== */

function displayReviews(){

    let container =
    document
    .getElementById(
        "reviewContainer"
    );


    container.innerHTML = "";


    let total =
    feedbacks.length;


    let totalRating = 0;


    feedbacks.forEach(
        item =>
        totalRating += item.rating
    );


    let average =
    total === 0
    ? 0
    : (totalRating / total)
        .toFixed(1);


    let positive =
    feedbacks.filter(
        item =>
        item.rating >= 4
    ).length;


    let suggestions =
    feedbacks.filter(
        item =>
        item.category === "Suggestion"
    ).length;


    document
    .getElementById(
        "reviewTotal"
    )
    .innerText =
    total;


    document
    .getElementById(
        "reviewAverage"
    )
    .innerText =
    average;


    document
    .getElementById(
        "reviewPositive"
    )
    .innerText =
    positive;


    document
    .getElementById(
        "reviewSuggestions"
    )
    .innerText =
    suggestions;


    if(feedbacks.length === 0){

        container.innerHTML = `

        <div
            style="
            grid-column:1/-1;
            text-align:center;
            padding:50px;
            background:white;
            border-radius:15px;
            "
        >

            <h2>
                No Feedback Yet
            </h2>

            <p>
                Be the first person to give feedback!
            </p>

        </div>

        `;

        return;

    }


    feedbacks
    .slice()
    .reverse()
    .forEach(
        function(item){

            let starsHTML =
            "★".repeat(item.rating)
            +
            "☆".repeat(
                5-item.rating
            );


            let firstLetter =
            item.name
            .charAt(0)
            .toUpperCase();


            container.innerHTML += `

            <div class="review-card">

                <div class="review-head">

                    <div class="avatar">
                        ${firstLetter}
                    </div>

                    <div>

                        <h3>
                            ${escapeHTML(item.name)}
                        </h3>

                        <div class="review-stars">
                            ${starsHTML}
                        </div>

                    </div>

                </div>


                <span class="category">

                    ${escapeHTML(item.category)}

                </span>


                <p>

                    "${escapeHTML(item.message)}"

                </p>


                <br>


                <small>

                    📅 ${item.date}

                </small>

            </div>

            `;

        }
    );

}

/* =====================================================
   ADMIN TABLE
===================================================== */

function displayAdminFeedback(){

    let table =
    document
    .getElementById(
        "feedbackTable"
    );


    let search =
    document
    .getElementById(
        "search"
    )
    .value
    .toLowerCase();


    let filter =
    document
    .getElementById(
        "filter"
    )
    .value;


    table.innerHTML = "";


    let result =
    feedbacks.filter(
        function(item){

            let searchMatch =

            item.name
            .toLowerCase()
            .includes(search)

            ||

            item.email
            .toLowerCase()
            .includes(search)

            ||

            item.message
            .toLowerCase()
            .includes(search);


            let categoryMatch =
            filter === "All"
            ||
            item.category === filter;


            return
            searchMatch
            &&
            categoryMatch;

        }
    );


    if(result.length === 0){

        table.innerHTML = `

        <tr>

            <td
                colspan="8"
                style="text-align:center"
            >

                No feedback found.

            </td>

        </tr>`;

        return;

    }


    result.forEach(
        function(item){

            let starsHTML =
            "★".repeat(
                item.rating
            );


            table.innerHTML += `

            <tr>

                <td>
                    ${item.id}
                </td>

                <td>
                    ${escapeHTML(item.name)}
                </td>

                <td>
                    ${escapeHTML(item.email)}
                </td>

                <td>
                    ${escapeHTML(item.category)}
                </td>

                <td style="color:#facc15">
                    ${starsHTML}
                </td>

                <td>
                    ${escapeHTML(item.message)}
                </td>

                <td>
                    ${item.date}
                </td>

                <td>

                    <button
                        class="delete"
                        onclick="deleteFeedback(${item.id})"
                    >

                        Delete

                    </button>

                </td>

            </tr>

            `;

        }
    );

}


/* =====================================================
   DELETE FEEDBACK
===================================================== */

function deleteFeedback(id){

    if(
        !confirm(
            "Are you sure you want to delete this feedback?"
        )
    ){

        return;

    }


    feedbacks =
    feedbacks.filter(
        item =>
        item.id !== id
    );


    localStorage.setItem(
        "feedbacks",
        JSON.stringify(feedbacks)
    );


    updateAdminDashboard();

    updateHome();


    showNotification(
        "Feedback deleted successfully!"
    );

}


/* =====================================================
   ANALYTICS
===================================================== */

function displayAnalytics(){

    let ratingBox =
    document
    .getElementById(
        "ratingAnalytics"
    );


    ratingBox.innerHTML = "";


    for(
        let rating=5;
        rating>=1;
        rating--
    ){

        let count =
        feedbacks.filter(
            item =>
            item.rating === rating
        ).length;


        let percentage =
        feedbacks.length === 0
        ? 0
        : (count /
           feedbacks.length) * 100;


        ratingBox.innerHTML += `

        <div class="bar">

            <div class="bar-top">

                <span>
                    ${rating} ⭐
                </span>

                <span>
                    ${count}
                </span>

            </div>

            <div class="bar-bg">

                <div
                    class="bar-fill"
                    style="width:${percentage}%"
                >

                </div>

            </div>

        </div>

        `;

    }


    let categoryBox =
    document
    .getElementById(
        "categoryAnalytics"
    );


    categoryBox.innerHTML = "";


    let categories = [

        "Product",
        "Service",
        "Website",
        "Customer Support",
        "Suggestion",
        "Other"

    ];


    categories.forEach(
        function(category){

            let count =
            feedbacks.filter(
                item =>
                item.category === category
            ).length;


            let percentage =
            feedbacks.length === 0
            ? 0
            : (count /
               feedbacks.length) * 100;


            categoryBox.innerHTML += `

            <div class="bar">

                <div class="bar-top">

                    <span>
                        ${category}
                    </span>

                    <span>
                        ${count}
                    </span>

                </div>

                <div class="bar-bg">

                    <div
                        class="bar-fill"
                        style="width:${percentage}%"
                    >

                    </div>

                </div>

            </div>

            `;

        }
    );

}

/* =====================================================
   NOTIFICATION
===================================================== */

function showNotification(message){

    let notification =
    document
    .getElementById(
        "notification"
    );


    notification.innerText =

/* =====================================================
   HTML SECURITY
===================================================== */

function escapeHTML(text){

    let div =
    document.createElement(
        "div"
    );


    div.textContent =
    text;


    return div.innerHTML;

}
